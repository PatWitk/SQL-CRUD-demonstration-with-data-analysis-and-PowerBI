    SELECTEDVALUE(products[product_name]),
    TOPN(1, ALL(products[product_name]), [total_units_sold], DESC)
)

CALCULATE(
    SELECTEDVALUE(products[category]),
    TOPN(1, ALL(products[product_name]), [total_units_sold], DESC)
)

[top_selling_product] & " (" & [top_selling_product_category] & ")"
