top_selling_product = 
CALCULATE(
    SELECTEDVALUE(products[product_name]),
    TOPN(1, ALL(products[product_name]), [total_units_sold], DESC)
)
/*This DAX code retrieves the name of of best selling product*/

top_selling_product_category = 
CALCULATE(
    SELECTEDVALUE(products[category]),
    TOPN(1, ALL(products[product_name]), [total_units_sold], DESC)
)
/*This DAX code retrieves the category name of of best selling producy*/

top_selling_product_display_card = 
[top_selling_product] & " (" & [top_selling_product_category] & ")"
/*This combines the previous name and category calculation into one combined display card*/
